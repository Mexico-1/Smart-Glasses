Core Components

Project Parts

Three main components working together to create an effective and reliable sensing system

Arduino Uno

The brain of the project. A microcontroller board that processes sensor signals and decides when to activate the buzzer based on measured distance.

ATmega328P 14 Digital 5V

HC-SR04 Sensor

Ultrasonic sensor that emits high-frequency sound waves and calculates distance based on the time it takes for the waves to bounce back from obstacles.

2cm - 400cm 40kHz 4 Pins

Buzzer

The alert buzzer. Emits audio signals at different rates depending on distance — the closer the obstacle, the faster the alert pulses.

Active 5V ~2kHz

// Smart Blind Glasses - Arduino Code
// Ultrasonic Sensor + Buzzer Alert System

const int trigPin = 9;
const int echoPin = 10;
const int buzzerPin = 8;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(buzzerPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  // Send ultrasonic pulse
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  // Read echo duration
  long duration = pulseIn(echoPin, HIGH);

  // Calculate distance in cm
  int distance = duration * 0.034 / 2;

  // Alert logic based on distance
  if (distance < 20) {
    // DANGER - Very fast alert
    digitalWrite(buzzerPin, HIGH);
    delay(50);
    digitalWrite(buzzerPin, LOW);
    delay(50);
  } else if (distance < 50) {
    // WARNING - Medium alert
    digitalWrite(buzzerPin, HIGH);
    delay(150);
    digitalWrite(buzzerPin, LOW);
    delay(150);
  } else if (distance < 100) {
    // CAUTION - Slow alert
    digitalWrite(buzzerPin, HIGH);
    delay(300);
    digitalWrite(buzzerPin, LOW);
    delay(300);
  } else {
    // SAFE - No alert
    digitalWrite(buzzerPin, LOW);
  }

  Serial.print("Distance: ");
  Serial.println(distance);
}
